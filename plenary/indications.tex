% #1: arrow rotation (0 = right, -45 = down-right, 90 = ahead, 180 = left).
\newcommand{\DirectionSign}[1]{%
  \null
  \begin{tikzpicture}[remember picture,overlay,x=1mm,y=1mm]
    \begin{scope}[shift={(current page.north west)}]
      \SignHeader

      \node[anchor=north west,inner sep=0pt,text=tgpetrol,
            font=\fontsize{64}{72}\selectfont\bfseries] at (12,-63)
        {HACKATHON};
      \node[anchor=north west,inner sep=0pt,text=tggraphit,
            font=\fontsize{42}{49}\selectfont] at (12,-94)
        {\HackathonRoom};

      \draw[tggrey,line width=0.4pt] (12,-122) -- (195,-122);

      \node[anchor=north west,inner sep=0pt,text=tgpetrol,
            font=\fontsize{64}{72}\selectfont\bfseries] at (12,-134)
        {PLENARY};
      \node[anchor=north west,inner sep=0pt,text=tggraphit,
            font=\fontsize{42}{49}\selectfont] at (12,-165)
        {\PlenaryRoom};

      % Solid vector arrow: legible from a distance and sharp in print.
      \begin{scope}[shift={(244,-128)},rotate=#1,scale=1.15]
        \fill[tgpetrol] (-31,-12) -- (0,-12) -- (0,-30)
          -- (34,0) -- (0,30) -- (0,12) -- (-31,12) -- cycle;
      \end{scope}
      \draw[tggrey,line width=0.4pt] (12,-198) -- (285,-198);
    \end{scope}
  \end{tikzpicture}%
  \newpage
}

% Full-page arrival sign: event name and room, without a direction arrow.
\newcommand{\ArrivalSign}[2]{%
  \null
  \begin{tikzpicture}[remember picture,overlay,x=1mm,y=1mm]
    \begin{scope}[shift={(current page.north west)}]
      \SignHeader
      \node[anchor=center,inner sep=0pt,text=tgpetrol,
            font=\fontsize{100}{110}\selectfont\bfseries] at (148.5,-94)
        {#1};
      \draw[tglime,line width=1.2pt] (70,-125) -- (227,-125);
      \node[anchor=center,inner sep=0pt,text=tggraphit,
            font=\fontsize{50}{58}\selectfont] at (148.5,-153)
        {#2};
      \draw[tggrey,line width=0.4pt] (12,-198) -- (285,-198);
    \end{scope}
  \end{tikzpicture}%
  \newpage
}

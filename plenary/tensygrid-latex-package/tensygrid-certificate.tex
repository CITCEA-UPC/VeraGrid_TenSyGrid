\newcommand{\Attendee}[1]{%
  \stepcounter{CertificateAttendeeCount}%
  \ifnum\CertificateOnly=0\relax
    \MakeAttendanceCertificate{#1}%
  \else
    \ifnum\CertificateOnly=\value{CertificateAttendeeCount}\relax
      \MakeAttendanceCertificate{#1}%
    \fi
  \fi}
\newcommand{\CertificateExportPass}{%
  \ShellEscape{echo 0 > certificates/\CertificateExportJob.export-status}%
  \ShellEscape{\CertificateExportCommand\space && echo 1 > certificates/\CertificateExportJob.export-status}%
  \openin\CertificateStatusRead=certificates/\CertificateExportJob.export-status\relax
  \read\CertificateStatusRead to \CertificateExportStatus
  \closein\CertificateStatusRead
  \ifnum\CertificateExportStatus=1\relax
  \else
    \CertificateExportSuccessfulfalse
    \PackageError{certificate}{Export failed for \CertificateExportJob}{Check the corresponding log file in certificates/. The individual PDF may be missing or out of date.}%
  \fi}
\newcommand{\CertificateExportAll}{%
  \setcounter{CertificateExportIndex}{1}%
  \loop\ifnum\value{CertificateExportIndex}<\numexpr\value{CertificateAttendeeCount}+1\relax
    \edef\CertificateExportJob{certificate-\ifnum\value{CertificateExportIndex}<10 00\else\ifnum\value{CertificateExportIndex}<100 0\fi\fi\arabic{CertificateExportIndex}}%
    \edef\CertificateExportCommand{xelatex -interaction=batchmode -halt-on-error -no-shell-escape -output-directory=certificates -jobname=\CertificateExportJob\space "\string\def\string\CertificateBatchExport{0}\string\def\string\CertificateOnly{\arabic{CertificateExportIndex}}\string\input{tensygrid-certificate.tex}"}%
    \CertificateExportSuccessfultrue
    \CertificateExportPass
    \ifCertificateExportSuccessful
      \CertificateExportPass
    \fi
    \ifCertificateExportSuccessful
      \typeout{Certificate exported: certificates/\CertificateExportJob.pdf}%
    \fi
    \stepcounter{CertificateExportIndex}%
  \repeat}

\begin{IEEEeqnarray}{rCl}
  \IEEEyesnumber\label{eq:ss}\IEEEyessubnumber*
  \frac{\mathrm{d}\mathbf{x}\left(t\right)}{\mathrm{d}t} & = & \mathbf{A}\mathbf{x}\left(t\right) + \mathbf{B}\mathbf{u}\left(t\right),\label{eq:ssx}\\
  y\left(t\right) & = & \mathbf{C}\mathbf{x}\left(t\right) + \mathbf{D}\mathbf{u}\left(t\right),\label{eq:ssy}
\end{IEEEeqnarray}
where~\eqref{eq:ss} contains the system evolution in~\eqref{eq:ssx} and the output in~\eqref{eq:ssy}.
